select distinct 
      p.program_code
    , b.short_name
	, b.vendor_internal_buyer_id
	, b.settlement_days_from_basis
	, c.short_name
	, c.gscf_company_id
	, vba.description
	, b.active_from_date
	, b.active_to_date
	, bdcc.type as configuration_type --todo later
from etb2.vendor_bank_account_mapping vbam
	inner join buyer b on b.buyer_id = vbam.buyer_id
	inner join vendor_ext_ident vei on vei.vendor_ext_ident_id = vbam.vendor_ext_ident_id
	inner join vendor v on v.vendor_id = vei.vendor_id
	inner join company c on c.company_id = v.company_id
	inner join etb2.vendor_bank_account vba on vba.id = vbam.vendor_bank_account_id
	inner join bank_account ba on ba.bank_account_id = vba.bank_account_id
	inner join etb2.basis_date_calculation_config bdcc on bdcc.program_id = b.program_id
	inner join etb2.basis_date_calculation_config_window bdccw on bdccw.basis_date_calculation_config_id = bdcc.id
	inner join program p on p.program_id = b.program_id
where p.program_code in ( 'DELAP', 'DELBR', 'DELEM', 'DELKB', 'DELKR', 'DELLA', 'DELNA' )
order by p.program_code, b.short_name;
