-- Buyer / Insured Limit
select distinct
	p.program_code
	, b.short_name as buyer
	, l.start_date
	, l.end_date
	, l.currency_id
	, c.short_name as insurer
	, isnull(l.limit_amount, 0) as 'amount'
	, ip.policy_name
from insured_limit il
	left join limit l on l.limit_id = il.limit_id
	inner join buyer b on b.buyer_id = il.buyer_id
	inner join program p on p.program_id = b.program_id
	inner join insurer i on il.insurer_id = i.insurer_id
	inner join company c on c.company_id = i.company_id
	left join etb2.insurer_policy ip on ip.id= il.insurer_policy_id
where p.program_code in ( 'DELAP', 'DELBR', 'DELEM', 'DELKB', 'DELKR', 'DELLA', 'DELNA' )
order by
	p.program_code,
	b.short_name;
