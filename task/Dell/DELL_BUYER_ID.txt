--buyer ids
select distinct
      p.program_code
    , b.short_name
	, bei.buyer_external_id
	, bei.valid_from
	, bei.valid_to
	, bei.reason
	, bei.customer_id
	, bei.settlement_days_from_basis
	, bei.payment_term
	from buyer_ext_ident bei
inner join buyer b on bei.buyer_id = b.buyer_id
inner join program p on p.program_id = b.program_id
where p.program_code in ( 'DELAP', 'DELBR', 'DELEM', 'DELKB', 'DELKR', 'DELLA', 'DELNA' )
order by p.program_code, b.short_name;
