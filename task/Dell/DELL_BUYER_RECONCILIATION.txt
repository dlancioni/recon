-- Buyer / Reconciliation
select distinct
      p.program_code
	, b.short_name
	, t.description
	, ba.bank_name
	, account_number
	, ba.currency_id
	, b.auto_reject_underpayments_below
	, b.auto_reject_overpayments_below
	, b.auto_reject_lpc_below
	, b.auto_reject_all_lpc
from funder_bank_account fba
	inner join funder f on f.funder_id = fba.funder_id
	inner join bank_account ba on ba.bank_account_id = fba.bank_account_id
	inner join program p on p.program_id = f.program_id
	inner join type t on t.type_id = fba.purpose_type_id
	inner join funder_ext_ident fei on fei.funder_id = f.funder_id
	inner join etb2.buyer_funder_entity bfe on bfe.funder_ext_ident_id = fei.funder_ext_ident_id
	inner join buyer b on b.buyer_id = bfe.buyer_id
where p.program_code in ( 'DELAP', 'DELBR', 'DELEM', 'DELKB', 'DELKR', 'DELLA', 'DELNA' )
order by p.program_code, short_name;
