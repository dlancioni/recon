### Important commands 
# build exe
pyinstaller.exe --onefile --icon=icon.ico --name recon main.py

# install dependencies
pip install -r requirements.txt

# test application
python -m unittest discover -stest -v

# run recon
python main.py --f "recon [en_us]" --r 0     (Sinthetic report)
python main.py --f "recon [en_us]" --r 1     (Analytic report side 1)
python main.py --f "recon [en_us]" --r 2     (Analytic report side 2)
python main.py --f "recon [en_us]" --r 12    (Analytic report) both sides
